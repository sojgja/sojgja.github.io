# Reset index
df_flat = df_multi.reset_index()

16. Style và Hiển Thị​

# Color gradient
df.style.background_gradient(cmap='Blues', subset=['salary'])

# Bar chart trong cell
df.style.bar(subset=['salary'], color='#66b0ff')

# Highlight max/min
df.style.highlight_max(subset=['salary'], color='lightgreen')
df.style.highlight_min(subset=['salary'], color='salmon')

# Format số
df.style.format({
    'salary': '${:,.0f}',
    'age': '{:.0f}',
    'pct': '{:.2%}'
})

# Conditional formatting
df.style.applymap(
    lambda x: 'color: red' if x > 90000 else 'color: black',
    subset=['salary']
)

17. Các Pattern và Tip Hay Dùng​

17.1 Method chaining​

result = (df
    .query('age >= 25')
    .assign(total=lambda x: x['salary'] + x['bonus'])
    .groupby('city')
    .agg(avg_total=('total', 'mean'))
    .reset_index()
    .sort_values('avg_total', ascending=False)
    .head(10)
)

17.2 Top N mỗi nhóm​

df.groupby('city').apply(
    lambda x: x.nlargest(3, 'salary')
).reset_index(drop=True)

17.3 Cumulative theo nhóm​

df['cum_sales'] = (df
    .sort_values(['product', 'date'])
    .groupby('product')['sales']
    .cumsum()
)

17.4 Shift trong nhóm​

df['prev_price'] = df.groupby('symbol')['price'].shift(1)
df['daily_return'] = df.groupby('symbol')['price'].pct_change()

17.5 Lag features cho ML​

for lag in [1, 3, 7, 30]:
    df[f'lag_{lag}'] = df.groupby('symbol')['price'].shift(lag)

Tổng kết​

Pandas là thư viện không thể thiếu cho bất kỳ ai làm việc với dữ liệu trong Python. Hãy nhớ:

  • Luôn dùng vectorization thay vì apply/loop
  • Downcast kiểu dữ liệu để tiết kiệm RAM
  • Dùng parquet/feather cho IO nhanh
  • Chunking khi làm việc với file lớn hơn RAM
  • Khi dữ liệu > 10GB, cân nhắc chuyển sang Polars
